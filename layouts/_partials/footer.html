{{- /* Identical on every page — baseof calls this through partialCached with no variant key, so it renders once per build regardless of corpus size. The three counts are O(1) slice/map lengths off structures Hugo has already built; nothing here walks site.RegularPages. */ -}} {{- /* Count notes, not every regular page — standalone pages like about.md and search.md are not notes. Each section's RegularPages is already built by Hugo, so this stays O(sections) rather than O(pages). */ -}} {{- $notes := 0 -}} {{- range site.Params.mainSections | default (slice "notes") -}} {{- with site.GetPage (printf "/%s" .) -}} {{- $notes = add $notes (len .RegularPages) -}} {{- end -}} {{- end -}} {{- $cats := len (index site.Taxonomies (site.Params.taxonomy.categoryPlural | default "categories")) -}} {{- $tags := len (index site.Taxonomies (site.Params.taxonomy.tagPlural | default "tags")) -}} {{- $blurb := site.Params.siteBlurb -}} {{- if not $blurb -}} {{- $blurb = printf "%s %s · %s %s · %s %s" (lang.FormatNumber 0 $notes) (cond (eq $notes 1) "note" "notes") (lang.FormatNumber 0 $cats) (cond (eq $cats 1) "category" "categories") (lang.FormatNumber 0 $tags) (cond (eq $tags 1) "tag" "tags") -}} {{- end -}}
