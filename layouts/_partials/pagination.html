{{- /* Server-rendered content pagination. Context: paginator — a Hugo paginator ceiling — optional cap on pages to link (0 = uncapped) Every control is a real to a paginator URL, so pages stay crawlable and shareable. Number windowing matches the sidebar mini-pager: page 1, current ±1, last, with … wherever the run skips. `ceiling` exists for the home view on very large corpora (decision D1 in PLAN.md): past it the pager stops and hands off to search rather than generating tens of thousands of pager pages. */ -}} {{- $paginator := .paginator -}} {{- $ceiling := int (.ceiling | default 0) -}} {{- $cur := $paginator.PageNumber -}} {{- $total := $paginator.TotalPages -}} {{- $capped := and (gt $ceiling 0) (gt $total $ceiling) -}} {{- if $capped }}{{ $total = $ceiling }}{{ end -}} {{- if gt $total 1 }} {{- $want := slice -}} {{- range (slice 1 (sub $cur 1) $cur (add $cur 1) $total) -}} {{- if and (ge . 1) (le . $total) -}}{{- $want = $want | append . -}}{{- end -}} {{- end -}} {{- $want = $want | uniq | sort -}}
{{- if $paginator.HasPrev }} ‹ Prev {{- else }} ‹ Prev {{- end }} {{- $prev := 0 }} {{- range $want }} {{- if gt (sub . $prev) 1 }} … {{- end }} {{- if eq . $cur }} {{ . }} {{- else }} {{ . }} {{- end }} {{- $prev = . }} {{- end }} {{- if and $paginator.HasNext (lt $cur $total) }} Next › {{- else }} Next › {{- end }}
{{- if $capped }}

Showing the first {{ lang.FormatNumber 0 (mul $total $paginator.PagerSize) }} of {{ lang.FormatNumber 0 $paginator.TotalNumberOfElements }} notes. Search to narrow them down.

{{- end }} {{- end }}
