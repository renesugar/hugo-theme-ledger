{{- /* The sidebar is byte-identical on every page, so baseof calls it through partialCached with no variant key: it renders once per build. The active row is marked client-side from location.pathname, which is what keeps it cacheable — see decision D3 in PLAN.md. Only the first page of each panel is inlined. The remaining terms live in one fingerprinted JSON asset shared by the whole site, so per-page HTML stays constant instead of growing with the number of terms — the reason a 50k-tag site does not ship 50k rows on every page. The asset is fetched once per visit (it also positions the panel on the active term) and is immutable, so it is served from cache for the rest of the session. */ -}} {{- $p := site.Params -}} {{- $limit := int ($p.taxonomyPageLimit | default 25) -}} {{- $maxTerms := int ($p.sidebar.maxTerms | default 200) -}} {{- $order := $p.sidebar.order | default "count" -}} {{- $catKey := $p.taxonomy.categoryPlural | default "categories" -}} {{- $tagKey := $p.taxonomy.tagPlural | default "tags" -}} {{- $catsPer := int ($p.pagination.sidebarCategories | default 7) -}} {{- $catsPerMobile := int ($p.pagination.sidebarCategoriesMobile | default 6) -}} {{- $tagsPer := int ($p.pagination.sidebarTags | default 9) -}} {{- $tagsPerMobile := int ($p.pagination.sidebarTagsMobile | default 8) -}} {{- $cats := partial "term-list.html" (dict "key" $catKey "limit" $limit "max" $maxTerms "order" $order "prefix" "category") -}} {{- $tags := partial "term-list.html" (dict "key" $tagKey "limit" $limit "max" $maxTerms "order" $order "prefix" "tag") -}} {{- /* Synthetic "All notes" row. The handoff's query grammar defines `category:All notes` as matching everything, and Home primes its search bar from the first sidebar category, so it belongs at the top of the list rather than being special-cased in the Home template. */ -}} {{- $allLabel := $p.sidebar.allNotesLabel | default "All notes" -}} {{- $catItems := $cats.terms -}} {{- $catTotal := $cats.total -}} {{- if $allLabel -}} {{- $notes := 0 -}} {{- range site.Params.mainSections | default (slice "notes") -}} {{- with site.GetPage (printf "/%s" .) -}}{{- $notes = add $notes (len .RegularPages) -}}{{- end -}} {{- end -}} {{- $allURL := printf "%s?q=%s" ("/search/" | relURL) (printf "category:%s" $allLabel | urlquery) -}} {{- $catItems = (slice (dict "name" $allLabel "url" $allURL "path" "" "count" $notes "over" (gt $notes $limit))) | append $catItems -}} {{- $catTotal = add $catTotal 1 -}} {{- end -}} {{- /* One shared, fingerprinted terms asset for client-side paging. */ -}} {{- $termsJSON := dict "categories" $catItems "tags" $tags.terms | jsonify | resources.FromString "ledger/terms.json" | fingerprint -}}
